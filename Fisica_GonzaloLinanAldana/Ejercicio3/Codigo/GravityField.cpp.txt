#include "GravityField.h"
#include "Components/ShapeComponent.h"
#include "Particle.h"
#include "RotatingParticle.h"

// Sets default values
AGravityField::AGravityField()
{
	PrimaryActorTick.bCanEverTick = true;
	fvForceDirection = FVector(0,0,-1);
	fForceIntensity = 100.0f;
}

void AGravityField::BeginPlay()
{
	Super::BeginPlay();
	GravityArea->OnComponentBeginOverlap.AddDynamic(this, &AGravityField::OnBeginOverlap);
	GravityArea->OnComponentEndOverlap.AddDynamic(this, &AGravityField::OnEndOverlap);
}

// Called every frame
void AGravityField::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);
}

//Casteamos a la particula para que detecte este actor como el "OtroActor"
void AGravityField::OnBeginOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor, UPrimitiveComponent* OtherComp, int32 OtherBodyIndex, bool bFromSweep, const FHitResult& SweepResult)
{
	AParticle* Particle = Cast<AParticle>(OtherActor);
	if (Particle)
	{
		// Aplicar fuerza lineal a las partículas normales
		FVector Force = fvForceDirection * fForceIntensity * Particle->GetMasa();
		Particle->SetAcceleration(Particle->GetAcceleration() + Force);
	}
	else
	{
		ARotatingParticle* RotatingParticle = Cast<ARotatingParticle>(OtherActor);
		if (RotatingParticle)
		{
			// Calcular y aplicar la fuerza angular a las partículas giratorias
			float AngularForceMagnitude = fForceIntensity * RotatingParticle->GetMasa();
			float MomentOfInertia = 2 * RotatingParticle->GetMasa() / 5;
			float AngularAcceleration = AngularForceMagnitude / MomentOfInertia;
			RotatingParticle->SetAngularAcceleration(RotatingParticle->GetAngularAcceleration() + AngularAcceleration);
		}
	}
}

void AGravityField::OnEndOverlap(UPrimitiveComponent* OverlappedComponent, AActor* OtherActor, UPrimitiveComponent* OtherComp, int32 OtherBodyIndex)
{
	//No utilizamos nada aqui

	
}
