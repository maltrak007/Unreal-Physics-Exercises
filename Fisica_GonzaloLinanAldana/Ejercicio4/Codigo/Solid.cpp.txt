#include "Solid.h"

// Sets default values
ASolid::ASolid()
{
 	// Set this actor to call Tick() every frame.  You can turn this off to improve performance if you don't need it.
	PrimaryActorTick.bCanEverTick = true;
	fvPosicion = FVector(140, 140, 0);
	fvVelocidad = FVector(100, 100, 0);
	fvAceleracion = FVector(0, 150, 0);
	fMasa = 1.0f;
}

// Called when the game starts or when spawned
void ASolid::BeginPlay()
{
	Super::BeginPlay();
}

// Called every frame
void ASolid::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);
	fvVelocidad += fvAceleracion * DeltaTime;
	fvPosicion += fvVelocidad * DeltaTime;
	SetActorLocation(fvPosicion);
}

FVector ASolid::GetPosition() const
{
	return fvPosicion;
}

void ASolid::SetPosition(const FVector& NewPosition)
{
	fvPosicion = NewPosition;
}

FVector ASolid::GetVelocity() const
{
	return fvVelocidad;
}

void ASolid::SetVelocity(const FVector& NewVelocity)
{
	fvVelocidad = NewVelocity;
}

FVector ASolid::GetAcceleration() const
{
	return fvAceleracion;
}

void ASolid::SetAcceleration(const FVector& NewAcceleration)
{
	fvAceleracion = NewAcceleration;
}

float ASolid::GetMasa() const
{
	return fMasa;
}

void ASolid::SetMasa(float NewMass)
{
	fMasa = NewMass;
}
